# Format: <ft_ls command> :: <ls model command>
# Example: ./ft_ls testdir :: ls -aU testdir | cat
#
# Remember: Columns without tty: -C | Colors without tty: --colors
#

./ft_ls                                                                      ::        ls -U
./ft_ls .                                                                    ::        ls -U .
./ft_ls Makefile                                                             ::        ls -U Makefile
./ft_ls testdir                                                              ::        ls -U testdir
./ft_ls ../../ft_ls/./test/testdir                                           ::        ls -U ../../ft_ls/./test/testdir
./ft_ls ../../ft_ls/./test/testdir/file                                      ::        ls -U ../../ft_ls/./test/testdir/file
./ft_ls /tmp/tempfile                                                        ::        ls -U /tmp/tempfile
./ft_ls /tmp/tempdir/                                                        ::        ls -U /tmp/tempdir/
./ft_ls Makefile run_e2e.sh Makefile generate_coverage_reports.sh            ::        ls -U Makefile run_e2e.sh Makefile generate_coverage_reports.sh
./ft_ls inc/ src testdir src                                                 ::        ls -U inc/ src testdir src
./ft_ls testdir/file2 testdir/firstDir testdir/secondDir testdir/file        ::        ls -U testdir/file2 testdir/firstDir testdir/secondDir testdir/file
./ft_ls testdir/empty                                                        ::        ls -U testdir/empty
./ft_ls testdir/.hidden2 testdir/.hidden                                     ::        ls -U testdir/.hidden2 testdir/.hidden
./ft_ls testdir/.hiddir2 testdir/.hiddir                                     ::        ls -U testdir/.hiddir2 testdir/.hiddir
./ft_ls testdir/.hidden2 testdir/.hiddir testdir/.hiddir2 testdir/.hidden    ::        ls -U testdir/.hidden2 testdir/.hiddir testdir/.hiddir2 testdir/.hidden
